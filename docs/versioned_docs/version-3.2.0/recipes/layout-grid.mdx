# Weighted layout grid

Available from Slackblocks **3.2.0**. Arrange service names and status text in two rows, using a 3:1 width ratio and an explicit linear fallback. The label is shown as bold text, not a heading. These examples construct a block; they do not send a message.

<RecipeOutput recipe="layout-grid" outputs={recipeOutputs} />

<LanguageContent>

<Python>

```python
from slackblocks import LayoutBlock, LayoutRow, LayoutItem, PlainText, SectionBlock


def section(text, identifier):
    return SectionBlock(text=PlainText(text), block_id=identifier)


layout = LayoutBlock(
    label="Service status", block_id="service-layout", is_label_visible=True,
    grid_lines="rows",
    rows=[
        LayoutRow(items=[
            LayoutItem(block=section("API", "api"), width="wide"),
            LayoutItem(block=section("Healthy", "api-status"), width="narrow", horizontal_alignment="end"),
        ]),
        LayoutRow(items=[
            LayoutItem(block=section("Worker", "worker"), width="wide"),
            LayoutItem(block=section("Delayed", "worker-status"), width="narrow", horizontal_alignment="end"),
        ]),
    ],
    fallback=[section("API: Healthy; Worker: Delayed", "status-fallback")],
)
print(layout)
```

```ts
import * as S from "@nicklambourne/slackblocks";

const section = (text: string, id: string) => S.SectionBlock()
  .text(S.PlainText().text(text)).blockId(id);
const row = (service: string, status: string, id: string) => S.LayoutRow().items([
  S.LayoutItem().block(section(service, id)).width("wide"),
  S.LayoutItem().block(section(status, `${id}-status`)).width("narrow").horizontalAlignment("end"),
]);
const layout = S.LayoutBlock().label("Service status").blockId("service-layout")
  .isLabelVisible(true).gridLines("rows")
  .rows([row("API", "Healthy", "api"), row("Worker", "Delayed", "worker")])
  .fallback([section("API: Healthy; Worker: Delayed", "status-fallback")]).build();
console.log(JSON.stringify(layout));
```

<Go>

```go
package main

import (
    "encoding/json"
    "os"
    s "github.com/nicklambourne/slackblocks/go/v3"
)

func section(text, id string) *s.SectionBlockBuilder {
    return s.NewSectionBlock().TextObject(s.NewPlainText().Text(text)).BlockID(id)
}
func row(service, status, id string) *s.LayoutRowBuilder {
    return s.NewLayoutRow().Items(
        s.NewLayoutItem().Block(section(service, id)).Width(s.LayoutWidthWide),
        s.NewLayoutItem().Block(section(status, id+"-status")).Width(s.LayoutWidthNarrow).HorizontalAlignment(s.LayoutAlignmentEnd),
    )
}
func main() {
    layout, err := s.NewLayoutBlock().Label("Service status").BlockID("service-layout").
        IsLabelVisible(true).GridLines(s.LayoutGridLinesRows).
        Rows(row("API", "Healthy", "api"), row("Worker", "Delayed", "worker")).
        Fallback(section("API: Healthy; Worker: Delayed", "status-fallback")).Build()
    if err != nil { panic(err) }
    if err := json.NewEncoder(os.Stdout).Encode(layout); err != nil { panic(err) }
}
```

```java
import io.github.nicklambourne.slackblocks.block.*;
import io.github.nicklambourne.slackblocks.object.*;

public final class Recipe {
  static SectionBlock section(String text, String id) {
    return SectionBlock.builder().text(PlainText.of(text)).blockId(id).build();
  }
  static LayoutRow row(String service, String status, String id) {
    return LayoutRow.builder().items(
        LayoutItem.builder().block(section(service, id)).width(LayoutWidth.WIDE).build(),
        LayoutItem.builder().block(section(status, id + "-status"))
            .width(LayoutWidth.NARROW).horizontalAlignment(LayoutAlignment.END).build()).build();
  }
  public static void main(String[] args) {
    var layout = io.github.nicklambourne.slackblocks.block.LayoutBlock.builder().label("Service status").blockId("service-layout")
        .isLabelVisible(true).gridLines(LayoutGridLines.ROWS)
        .rows(row("API", "Healthy", "api"), row("Worker", "Delayed", "worker"))
        .fallback(section("API: Healthy; Worker: Delayed", "status-fallback")).build();
    System.out.println(layout.toJson());
  }
}
```

```rust
use slackblocks::*;

fn section(text: &str, id: &str) -> Result<SectionBlock, ValidationError> {
    SectionBlock::builder().text(PlainText::new(text)?).block_id(id).build()
}
fn row(service: &str, status: &str, id: &str) -> Result<LayoutRow, ValidationError> {
    LayoutRow::builder().items([
        LayoutItem::builder().block(section(service, id)?).width(LayoutWidth::Wide).build()?,
        LayoutItem::builder().block(section(status, &format!("{id}-status"))?)
            .width(LayoutWidth::Narrow).horizontal_alignment(LayoutAlignment::End).build()?,
    ]).build()
}
fn main() -> Result<(), Box<dyn std::error::Error>> {
    let layout = LayoutBlock::builder().label("Service status").block_id("service-layout")
        .is_label_visible(true).grid_lines(LayoutGridLines::Rows)
        .rows([row("API", "Healthy", "api")?, row("Worker", "Delayed", "worker")?])
        .fallback([LayoutFallback::from(section("API: Healthy; Worker: Delayed", "status-fallback")?)])
        .build()?;
    println!("{}", serde_json::to_string(&layout)?);
    Ok(())
}
```

## Layout and fallback behavior

- Rows contain one to three items; a layout has one to five rows. `narrow`, `standard` and `wide` have weights 1, 2 and 3. An omitted width is standard; omitted alignment fills the cell.
- `rows` draws the outer border and horizontal row separators. `outer` draws only the border; `none` draws neither. There are no vertical grid lines.
- Fallback is a separate linear representation for clients that cannot render layouts. The library preserves omission and explicit empty arrays; it does not invent a fallback. Put essential information in the message's top-level text too.
- Hidden labels are not accessible names. Preserve meaningful source order, do not convey status through colour alone, and test your app with assistive technology. Slack may wrap cells on narrow clients; this local preview is illustrative.

See [layout constraints](/usage/using_blocks#layout-grids) and [validation boundaries](/usage/validation-boundary#layout-accounting) before composing larger grids. Older-client fallback and mobile/screen-reader behavior are documented contracts, not guarantees established by this preview.
