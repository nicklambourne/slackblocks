# Build status notification

Summarize a build with a plain-text notification fallback, status fields and contextual detail.

<RecipeOutput recipe="build-notification" outputs={recipeOutputs} />

## Code

<LanguageContent>

<Python>

A CI notification with summary fields and a contextual footer.


```python
from slackblocks import (
    ContextBlock,
    DividerBlock,
    HeaderBlock,
    Message,
    SectionBlock,
    Text,
)


def build_status_message(
    *,
    channel: str,
    project: str,
    branch: str,
    commit: str,
    author: str,
    duration: str,
    passed: bool,
) -> Message:
    icon = ":white_check_mark:" if passed else ":x:"
    status = "passed" if passed else "failed"

    return Message(
        channel=channel,
        text=f"Build {status} for {project}@{branch}",  # plain-text fallback
        blocks=[
            HeaderBlock(f"{icon} {project} build {status}"),
            SectionBlock(
                fields=[
                    f"*Branch*\n`{branch}`",
                    f"*Commit*\n`{commit[:7]}`",
                    f"*Author*\n{author}",
                    f"*Duration*\n{duration}",
                ],
            ),
            DividerBlock(),
            ContextBlock(
                elements=[
                    Text(f":calendar: Triggered by push to `{branch}`"),
                ],
            ),
        ],
    )
print(build_status_message(channel="C1", project="example", branch="main", commit="abc1234", author="Nick", duration="3m 12s", passed=True).json())
```





Tips:

- Always pass a `text=` fallback to `Message` — Slack uses it for desktop and mobile push notifications.
- `SectionBlock` accepts up to 10 `fields`, each up to 2,000 characters.

```ts
import {
  ContextBlock,
  DividerBlock,
  HeaderBlock,
  Markdown,
  Message,
  SectionBlock,
} from "@nicklambourne/slackblocks";

function buildStatusMessage(input: {
  channel: string;
  project: string;
  branch: string;
  commit: string;
  author: string;
  duration: string;
  passed: boolean;
}) {
  const icon = input.passed ? ":white_check_mark:" : ":x:";
  const status = input.passed ? "passed" : "failed";

  return Message()
    .channel(input.channel)
    .text(`Build ${status} for ${input.project}@${input.branch}`)
    .blocks(
      HeaderBlock().text(`${icon} ${input.project} build ${status}`),
      SectionBlock().fields(
        `*Branch*\n\`${input.branch}\``,
        `*Commit*\n\`${input.commit.slice(0, 7)}\``,
        `*Author*\n${input.author}`,
        `*Duration*\n${input.duration}`,
      ),
      DividerBlock(),
      ContextBlock().elements(
        Markdown().text(`:calendar: Triggered by push to \`${input.branch}\``),
      ),
    )
    .build();
}
console.log(JSON.stringify(buildStatusMessage({channel: "C1", project: "example", branch: "main", commit: "abc1234", author: "Nick", duration: "3m 12s", passed: true})));
```

```go
package main

import (
	"encoding/json"
	"fmt"
	"os"

	slackblocks "github.com/nicklambourne/slackblocks/go/v3"
)

func BuildStatusMessage(channel, project, branch, commit, author, duration string, passed bool) (slackblocks.Object, error) {
	icon, status := ":x:", "failed"
	if passed {
		icon, status = ":white_check_mark:", "passed"
	}

	return slackblocks.NewMessage().
		Channel(channel).
		Text(fmt.Sprintf("Build %s for %s@%s", status, project, branch)).
		Blocks(
			slackblocks.NewHeaderBlock().Text(fmt.Sprintf("%s %s build %s", icon, project, status)),
			slackblocks.NewSectionBlock().Fields(
				fmt.Sprintf("*Branch*\n`%s`", branch),
				fmt.Sprintf("*Commit*\n`%.7s`", commit),
				fmt.Sprintf("*Author*\n%s", author),
				fmt.Sprintf("*Duration*\n%s", duration),
			),
			slackblocks.NewDividerBlock(),
			slackblocks.NewContextBlock().Elements(
				slackblocks.NewMarkdown().Text(fmt.Sprintf(":calendar: Triggered by push to `%s`", branch)),
			),
		).
		Build()
}

func main() {
	value, err := BuildStatusMessage("C1", "example", "main", "abc1234", "Nick", "3m 12s", true)
	if err != nil {
		panic(err)
	}
	if err := json.NewEncoder(os.Stdout).Encode(value); err != nil {
		panic(err)
	}
}
```

```java
import io.github.nicklambourne.slackblocks.block.ContextBlock;
import io.github.nicklambourne.slackblocks.block.DividerBlock;
import io.github.nicklambourne.slackblocks.block.HeaderBlock;
import io.github.nicklambourne.slackblocks.block.SectionBlock;
import io.github.nicklambourne.slackblocks.object.MarkdownText;
import io.github.nicklambourne.slackblocks.payload.MessagePayload;

public final class Recipe {

  static MessagePayload buildStatusMessage(
  String channel,
  String project,
  String branch,
  String commit,
  String author,
  String duration,
  boolean passed) {
    String icon = passed ? ":white_check_mark:" : ":x:";
    String status = passed ? "passed" : "failed";

    return MessagePayload.builder()
    .channel(channel)
    .text("Build " + status + " for " + project + "@" + branch)
    .blocks(
    HeaderBlock.builder().text(icon + " " + project + " build " + status).build(),
    SectionBlock.builder().markdownFields(
    "*Branch*\n`" + branch + "`",
    "*Commit*\n`" + commit.substring(0, Math.min(7, commit.length())) + "`",
    "*Author*\n" + author,
    "*Duration*\n" + duration).build(),
    DividerBlock.builder().build(),
    ContextBlock.builder().elements(
    MarkdownText.of(":calendar: Triggered by push to `" + branch + "`")).build())
    .build();
  }
  public static void main(String[] args) {
    System.out.println(buildStatusMessage("C1", "example", "main", "abc1234", "Nick", "3m 12s", true).toJson());
  }
}
```

```haskell
{-# LANGUAGE OverloadedStrings #-}

import Data.Aeson (encode)
import qualified Data.Map.Strict as Map
import Slackblocks

example :: Result MessagePayload
example = do
  title <- buildPlainText (plainText "Build #482 passed")
  heading <- buildHeaderBlock (headerBlock title)
  branch <- buildMarkdownText (markdownText "*Branch*\n`main`")
  tests <- buildMarkdownText (markdownText "*Tests*\n1,247 passed")
  details <-
    buildSectionBlock
      ( sectionBlock
          { sectionBlockFields = Just (map markdownTextAsText [branch, tests])
          }
      )
  label <- buildPlainText (plainText "View build")
  button <-
    buildButtonElement
      ( (buttonElement label)
          { buttonElementActionId = Just "view",
            buttonElementUrl = Just "https://ci.example.com/482"
          }
      )
  actions <- buildActionsBlock (actionsBlock [buttonElementAsElement button])
  buildMessagePayload
    ( (messagePayload "C0123456789")
        { messagePayloadText = Just "Build #482 passed",
          messagePayloadBlocks = Just [headerBlockAsBlock heading, sectionBlockAsBlock details, actionsBlockAsBlock actions]
        }
    )
```
