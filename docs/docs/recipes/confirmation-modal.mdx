# Confirmation modal

Construct a view with an input and submit button. Opening it requires a genuine interaction trigger ID; processing the submission belongs to your application.

<RecipeOutput recipe="confirmation-modal" outputs={recipeOutputs} />

## Code

<LanguageContent>

<Python>

Pass the constructed modal to `views.open` with an interaction trigger ID. The example only constructs the view; it makes no Slack request.

```python
from slackblocks import (
    InputBlock,
    Modal,
    PlainTextInput,
    SectionBlock,
)


def confirmation_modal(target: str) -> Modal:
    modal = Modal(
        title="Confirm deletion",
        submit="Delete",
        close="Cancel",
        callback_id="confirm_delete",
        private_metadata=target,  # round-tripped back to your handler on submit
        blocks=[
            SectionBlock(f":warning: This will permanently delete *{target}*."),
            InputBlock(
                label="Type the name to confirm",
                element=PlainTextInput(
                    action_id="confirmation_text",
                    placeholder=target,
                ),
                block_id="confirmation",
            ),
        ],
    )

    return modal

print(confirmation_modal("production").json())
```

package main

import (

func main() {
	value, err := s.NewModal().Title("Confirm deployment").Submit("Deploy").CallbackID("deploy").Blocks(
		s.NewInputBlock().Label("Reason").BlockID("reason").Element(s.NewPlainTextInput().ActionID("reason_input")),
	).Build()

		panic(err)
	}

		panic(err)
	}
}

use slackblocks::*;

fn main() -> Result<(), Box<dyn std::error::Error>> {
    let modal = ModalView::builder()
        .title("Approval")
        .submit("Approve")
        .close("Cancel")
        .block(
            InputBlock::builder()
                .label("Reason")
                .element(
                    PlainTextInputElement::builder()
                        .action_id("reason")
                        .build()?,
                )
                .build()?,
        )
        .build()?;
    println!("{}", serde_json::to_string(&modal)?);
    Ok(())
}

import Data.Aeson (encode)

import Slackblocks

example = do
  title <- buildPlainText (plainText "Confirm deployment")
  submit <- buildPlainText (plainText "Deploy")
  label <- buildPlainText (plainText "Reason")
  field <- buildPlainTextInputElement (plainTextInputElement {plainTextInputElementActionId = Just "reason_input"})
  input <- buildInputBlock ((inputBlock label (plainTextInputElementAsInputElement field)) {inputBlockBlockId = Just "reason"})
  buildModalView ((modalView title [inputBlockAsBlock input]) {modalViewSubmit = Just submit, modalViewCallbackId = Just "deploy"})
