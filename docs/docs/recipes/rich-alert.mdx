# Rich-formatted alert

Use native rich-text values to emphasize an operational notice without constructing raw Block Kit JSON.

<RecipeOutput recipe="rich-alert" outputs={recipeOutputs} />

## Code

<LanguageContent>

<Python>

Use `RichTextBlock` when you want inline bold/italic/code styling without writing markdown by hand.

```python
from slackblocks import (
    Message,
    RichText,
    RichTextBlock,
    RichTextLink,
    RichTextSection,
)


def deploy_alert(*, channel: str, service: str, version: str, dashboard_url: str) -> Message:
    return Message(
        channel=channel,
        text=f"Deployed {service} {version}",
        blocks=[
            RichTextBlock(
                RichTextSection(
                    elements=[
                        RichText(text="Deployed "),
                        RichText(text=service, bold=True),
                        RichText(text=" version "),
                        RichText(text=version, code=True),
                        RichText(text=". "),
                        RichTextLink(url=dashboard_url, text="View dashboard"),
                        RichText(text="."),
                    ],
                ),
            ),
        ],
    )
print(deploy_alert(channel="C1", service="api", version="3.1.0", dashboard_url="https://status.example.com").json())
```

<Go>

```go
package main

import (
	"encoding/json"
	s "github.com/nicklambourne/slackblocks/go/v3"
	"os"
)

func main() {
	value, err := s.NewMessage().Channel("C1").Text("Deployment needs attention").Blocks(
		s.NewRichTextBlock().Elements(s.NewRichTextSection().Elements(
			s.NewRichText().Text("Deployment needs attention").Style(s.RichTextStyle{"bold": true}),
		)),
	).Build()
	if err != nil {
		panic(err)
	}
	if err := json.NewEncoder(os.Stdout).Encode(value); err != nil {
		panic(err)
	}
}
```

```rust
use slackblocks::*;

fn main() -> Result<(), Box<dyn std::error::Error>> {
    let rich = RichTextBlock::builder()
        .element(
            RichTextSection::builder()
                .element(
                    RichTextText::builder()
                        .text("Deployment needs attention")
                        .style(RichTextStyle::new().bold(true))
                        .build()?,
                )
                .build()?,
        )
        .build()?;
    let message = MessagePayload::builder("C01234567")
        .text("Deployment needs attention")
        .block(rich)
        .build()?;
    println!("{}", serde_json::to_string(&message)?);
    Ok(())
}
```

<Haskell>

```haskell
{-# LANGUAGE OverloadedStrings #-}

import Data.Aeson (encode)
import qualified Data.Map.Strict as Map
import Slackblocks

example = do
  title <- buildPlainText (plainText "Deployment")
  warning <- buildMarkdownText (markdownText "*Attention:* deployment paused")
  alert <- buildAlertBlock ((alertBlock (markdownTextAsText warning)) {alertBlockLevel = Just AlertLevelWarning})
  text <- buildRichTextText ((richTextText "Review the checks") {richTextTextStyle = Just (richTextStyle {styleBold = Just True})})
  section <- buildRichTextSection (richTextSection [richTextTextAsRichTextSectionElement text])
  rich <- buildRichTextBlock (richTextBlock [richTextSectionAsRichTextBlockElement section])
  buildModalView (modalView title [alertBlockAsBlock alert, richTextBlockAsBlock rich])
```
